cmake_minimum_required(VERSION 3.15)
project(tlbx_connect_native_load C CXX)
add_executable(connect-native-load load.c)
target_link_libraries(connect-native-load PRIVATE ${CMAKE_DL_LIBS})
set_property(TARGET connect-native-load PROPERTY C_STANDARD 11)
if(MSVC)
    set_property(TARGET connect-native-load PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
if(NATIVE_DIRECTORY AND NATIVE_PREFIX)
    add_library(connect-runtime SHARED IMPORTED)
    if(WIN32)
        set_target_properties(connect-runtime PROPERTIES IMPORTED_LOCATION "${NATIVE_DIRECTORY}/datachannel.dll" IMPORTED_IMPLIB "${NATIVE_PREFIX}/lib/datachannel.lib")
    elseif(APPLE)
        set_target_properties(connect-runtime PROPERTIES IMPORTED_LOCATION "${NATIVE_DIRECTORY}/libdatachannel.dylib")
    else()
        set_target_properties(connect-runtime PROPERTIES IMPORTED_LOCATION "${NATIVE_DIRECTORY}/libdatachannel.so")
    endif()
    add_executable(connect-native-parse parse.cpp)
    target_include_directories(connect-native-parse PRIVATE "${NATIVE_PREFIX}/include")
    target_link_libraries(connect-native-parse PRIVATE connect-runtime)
    if(APPLE)
        set_property(TARGET connect-native-parse PROPERTY BUILD_RPATH "@loader_path")
    elseif(UNIX)
        set_property(TARGET connect-native-parse PROPERTY BUILD_RPATH "$ORIGIN")
    endif()
    set_property(TARGET connect-native-parse PROPERTY CXX_STANDARD 17)
    if(MSVC)
        set_property(TARGET connect-native-parse PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    endif()
endif()
